<?php
header('Content-Type: application/json');

// Check if the form was submitted using the POST method
if ($_SERVER["REQUEST_METHOD"] == "POST") {

    // Set the recipient email address. THIS IS WHERE YOU PUT YOUR EMAIL.
    $to = "amirwigodz@gmail.com";

    // Get the feedback message from the form data
    $feedback = htmlspecialchars(trim($_POST['feedback']));

    // Get the sender's email (optional)
    $from_email = htmlspecialchars(trim($_POST['email']));
    
    // Set the subject line for the email
    $subject = "New Feedback from Crossword Website";

    // Build the email message body
    $email_body = "You have received new feedback from your crossword website.\n\n";
    $email_body .= "Feedback: " . $feedback . "\n";
    
    // Add the sender's email to the body if it was provided
    if (!empty($from_email)) {
        $email_body .= "Sender's Email: " . $from_email;
    }

    // Set additional headers
    $headers = "From: webmaster@yourwebsite.com"; // Change to your website's domain or a no-reply email
    $headers .= "\r\nReply-To: " . (empty($from_email) ? 'noreply@yourwebsite.com' : $from_email);

    // Send the email
    if (mail($to, $subject, $email_body, $headers)) {
        // Return a success message
        echo json_encode(["status" => "success", "message" => "Thank you! Your feedback has been sent."]);
    } else {
        // Return an error message
        echo json_encode(["status" => "error", "message" => "Oops! Something went wrong. Please try again later."]);
    }
} else {
    // If the request method is not POST, return an error
    echo json_encode(["status" => "error", "message" => "Invalid request method."]);
}
?>
